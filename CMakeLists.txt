cmake_minimum_required(VERSION 3.20)
project(janus_noise VERSION 0.1.0 LANGUAGES CXX)

add_library(janus_core core/src/config.cpp core/src/game.cpp core/src/replay.cpp)
target_include_directories(janus_core PUBLIC core/include)
target_compile_features(janus_core PUBLIC cxx_std_20)
set_target_properties(janus_core PROPERTIES CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
add_library(janus_bots bots/src/bots.cpp)
target_include_directories(janus_bots PUBLIC bots/include)
target_link_libraries(janus_bots PUBLIC janus_core)
add_library(janus_simulator simulator/src/simulator.cpp)
target_include_directories(janus_simulator PUBLIC simulator/include)
target_link_libraries(janus_simulator PUBLIC janus_bots Threads::Threads)
add_executable(janus_cli simulator/src/main.cpp)
target_link_libraries(janus_cli PRIVATE janus_simulator)

set_target_properties(janus_bots janus_simulator janus_cli PROPERTIES CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
  add_executable(contract_checks tests/contract_checks.cpp)
  target_link_libraries(contract_checks PRIVATE janus_core)
  add_test(NAME core_contracts COMMAND contract_checks)
  add_executable(game_checks tests/game_checks.cpp)
  target_link_libraries(game_checks PRIVATE janus_core)
  target_include_directories(game_checks PRIVATE core/src)
  target_compile_definitions(game_checks PRIVATE FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/replay-v1.json")
  add_test(NAME core_gameplay COMMAND game_checks)
  add_executable(simulator_checks tests/simulator_checks.cpp)
  target_link_libraries(simulator_checks PRIVATE janus_simulator)
  add_test(NAME simulator_bots COMMAND simulator_checks)
  add_test(NAME cli_match COMMAND janus_cli match --seed 42 --first heuristic --second random --save "${CMAKE_CURRENT_BINARY_DIR}/cli-replay.json")
  add_test(NAME cli_replay COMMAND janus_cli replay --file "${CMAKE_CURRENT_BINARY_DIR}/cli-replay.json")
  set_tests_properties(cli_match PROPERTIES FIXTURES_SETUP cli_replay_file)
  set_tests_properties(cli_replay PROPERTIES FIXTURES_REQUIRED cli_replay_file)
  add_test(NAME cli_human_play COMMAND ${CMAKE_COMMAND} -DCLI=$<TARGET_FILE:janus_cli> -DWORK=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_play.cmake)
  add_test(NAME cli_odd_evaluation COMMAND janus_cli evaluate --games 3)
  set_tests_properties(cli_odd_evaluation PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_invalid_workers COMMAND janus_cli batch --workers 0)
  set_tests_properties(cli_invalid_workers PROPERTIES WILL_FAIL TRUE)
endif()
